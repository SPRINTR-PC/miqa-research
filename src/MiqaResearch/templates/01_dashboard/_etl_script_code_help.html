Python

{% translate "main returns or yields one dict per row, with the datatype's field names as keys." %}

import csv

def main(ctx):
    delimiter = ctx.config.get("delimiter", ",")
    for path in sorted(ctx.input_dir.glob("*.csv")):
        with path.open(newline="", encoding="utf-8-sig") as file:
            for record in csv.DictReader(file, delimiter=delimiter):
                yield {
                    "person_source_value": record["Personnummer"],
                    # ... the other fields of the datatype
                }

R

{% translate "main calls emit(row) once per row, where row is a named list (or a one-row data.frame) with the datatype's field names. jsonlite is available." %}

main <- function(ctx) {
  delimiter <- if (is.null(ctx$config$delimiter)) "," else ctx$config$delimiter
  for (path in list.files(ctx$input_dir, pattern = "\\.csv$", full.names = TRUE)) {
    data <- read.csv(path, sep = delimiter, check.names = FALSE)
    for (i in seq_len(nrow(data))) {
      emit(list(
        person_source_value = data[["Personnummer"]][i]
        # ... the other fields of the datatype
      ))
    }
  }
}
